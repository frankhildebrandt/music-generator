cmake_minimum_required(VERSION 3.24)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
project(MusicGeneratorLoops VERSION 0.1.2 LANGUAGES C CXX)
if(APPLE AND NOT CMAKE_OSX_DEPLOYMENT_TARGET)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.3")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CACHE_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.cache" CACHE PATH "Pinned plugin dependency cache")
set(IPLUG2_DIR "${CACHE_ROOT}/iplug2" CACHE PATH "iPlug2 checkout")
add_library(loop_dsp STATIC src/Audio.cpp src/Bank.cpp src/Player.cpp)
target_include_directories(loop_dsp PUBLIC src "${CACHE_ROOT}/json" "${CACHE_ROOT}/signalsmith-stretch" "${CACHE_ROOT}/signalsmith-linear/include")
target_compile_options(loop_dsp PRIVATE $<$<CXX_COMPILER_ID:AppleClang,Clang,GNU>:-Wall;-Wextra;-Wpedantic>)
option(LOOPS_DSP_ONLY "Build portable DSP tests only" OFF)
include(CTest)
add_executable(loop_tests tests/player.cpp)
target_link_libraries(loop_tests PRIVATE loop_dsp)
add_test(NAME loop_dsp_contracts COMMAND loop_tests)
add_executable(realtime_tests tests/realtime.cpp)
target_link_libraries(realtime_tests PRIVATE loop_dsp)
add_test(NAME loop_realtime_contracts COMMAND realtime_tests)
add_executable(timing_tests tests/timing.cpp)
target_link_libraries(timing_tests PRIVATE loop_dsp)
add_test(NAME loop_timing_contracts COMMAND timing_tests)
if(NOT LOOPS_DSP_ONLY)
  if(UNIX)
    add_executable(clap_host tests/clap_host.cpp)
    target_link_libraries(clap_host PRIVATE loop_dsp)
    target_include_directories(clap_host PRIVATE "${IPLUG2_DIR}/Dependencies/IPlug/CLAP_SDK/include")
  endif()
  if(WIN32)
    enable_language(RC)
  endif()
  include(${IPLUG2_DIR}/iPlug2.cmake)
  find_package(iPlug2 REQUIRED)
  iplug_add_plugin(MusicGeneratorLoops SOURCES src/MusicGeneratorLoops.cpp src/MusicGeneratorLoops.h src/Editor.cpp src/EditorViews.cpp src/GenerationView.cpp src/Takes.cpp src/Controller.cpp src/Service.cpp resources/resource.h RESOURCES resources/Roboto-Regular.ttf FORMATS APP VST3 CLAP LINK loop_dsp)
endif()

if(WIN32 AND NOT LOOPS_DSP_ONLY)
  foreach(format vst3 clap)
    target_sources(MusicGeneratorLoops-${format} PRIVATE resources/plugin.rc)
    target_include_directories(MusicGeneratorLoops-${format} PRIVATE resources)
  endforeach()
endif()
